local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")

local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

--------------------------------------------------
-- CONFIGURATION
--------------------------------------------------

local HEALTH_CONFIG = {
	Width = 5,
	Gap = 6,

	BackgroundColor = Color3.fromRGB(12, 12, 12),
	BackgroundTransparency = 0.15,

	BorderThickness = 1,

	DamageDropTime = 0.30,

	TopColor = Color3.fromRGB(50, 255, 50),
	MiddleColor = Color3.fromRGB(255, 255, 0),
	BottomColor = Color3.fromRGB(255, 35, 35),

	OutlineWhite = Color3.fromRGB(255, 255, 255),
	OutlineBlack = Color3.fromRGB(0, 0, 0),

	OutlineAnimationTime = 1.0,

	GlowEnabled = true,
	GlowTransparency = 0.60,
	MaxDistance = 2000,
}

--------------------------------------------------
-- CORE INTERFACE & STORAGE
--------------------------------------------------

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "Standalone_Health_ESP"
ScreenGui.IgnoreGuiInset = true
ScreenGui.ResetOnSpawn = false
ScreenGui.DisplayOrder = 999999
ScreenGui.Parent = game:GetService("CoreGui")

local HealthESPObjects = {}
local Enabled = true

--------------------------------------------------
-- HELPERS & BOUNDS
--------------------------------------------------

local function getOutlineRotation()
	local speed = 180 / math.max(HEALTH_CONFIG.OutlineAnimationTime, 0.05)
	return 90 + (os.clock() * speed) % 360
end

local function getBounds(character)
	local cf, size = character:GetBoundingBox()
	local half = size / 2

	local points = {
		cf * Vector3.new(-half.X, -half.Y, -half.Z),
		cf * Vector3.new(-half.X, -half.Y, half.Z),
		cf * Vector3.new(-half.X, half.Y, -half.Z),
		cf * Vector3.new(-half.X, half.Y, half.Z),
		cf * Vector3.new(half.X, -half.Y, -half.Z),
		cf * Vector3.new(half.X, -half.Y, half.Z),
		cf * Vector3.new(half.X, half.Y, -half.Z),
		cf * Vector3.new(half.X, half.Y, half.Z)
	}

	local minX, minY = math.huge, math.huge
	local maxX, maxY = -math.huge, -math.huge
	local visible = false

	for _, point in ipairs(points) do
		local screenPosition, onScreen = Camera:WorldToViewportPoint(point)

		if screenPosition.Z <= 0 then
			return nil
		end

		if onScreen then
			visible = true
		end

		minX = math.min(minX, screenPosition.X)
		minY = math.min(minY, screenPosition.Y)
		maxX = math.max(maxX, screenPosition.X)
		maxY = math.max(maxY, screenPosition.Y)
	end

	if not visible then
		return nil
	end

	return minX, minY, maxX, maxY
end

local function getCharacterData(player)
	local character = player.Character
	if not character then return nil end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")

	if not humanoid or humanoid.Health <= 0 or not root then
		return nil
	end

	local distance = (Camera.CFrame.Position - root.Position).Magnitude
	if distance > HEALTH_CONFIG.MaxDistance then
		return nil
	end

	local minX, minY, maxX, maxY = getBounds(character)
	if not minX then return nil end

	local width = maxX - minX
	local height = maxY - minY

	if width < 3 or height < 3 then return nil end

	local scale = math.clamp(math.min(width, height) / 45, 0.3, 1)

	return {
		minX = minX,
		minY = minY,
		maxX = maxX,
		maxY = maxY,
		scale = scale,
		humanoid = humanoid
	}
end

--------------------------------------------------
-- HEALTH OBJECT LIFECYCLE
--------------------------------------------------

local function createHealthESP(player)
	if HealthESPObjects[player] then
		return HealthESPObjects[player]
	end

	local object = {}

	object.container = Instance.new("Frame")
	object.container.Name = "Health_" .. player.Name
	object.container.BackgroundTransparency = 1
	object.container.BorderSizePixel = 0
	object.container.Visible = false
	object.container.ZIndex = 20
	object.container.Parent = ScreenGui

	object.glow = Instance.new("Frame")
	object.glow.Name = "HealthGlow_" .. player.Name
	object.glow.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	object.glow.BackgroundTransparency = HEALTH_CONFIG.GlowTransparency
	object.glow.BorderSizePixel = 0
	object.glow.Visible = false
	object.glow.ZIndex = 19
	object.glow.Parent = ScreenGui

	object.background = Instance.new("Frame")
	object.background.BackgroundColor3 = HEALTH_CONFIG.BackgroundColor
	object.background.BackgroundTransparency = HEALTH_CONFIG.BackgroundTransparency
	object.background.BorderSizePixel = 0
	object.background.ClipsDescendants = true
	object.background.Size = UDim2.fromScale(1, 1)
	object.background.ZIndex = 20
	object.background.Parent = object.container

	object.gradientBar = Instance.new("Frame")
	object.gradientBar.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	object.gradientBar.BorderSizePixel = 0
	object.gradientBar.Size = UDim2.fromScale(1, 1)
	object.gradientBar.ZIndex = 21
	object.gradientBar.Parent = object.background

	local gradient = Instance.new("UIGradient")
	gradient.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, HEALTH_CONFIG.TopColor),
		ColorSequenceKeypoint.new(0.5, HEALTH_CONFIG.MiddleColor),
		ColorSequenceKeypoint.new(1, HEALTH_CONFIG.BottomColor),
	})
	gradient.Rotation = 90
	gradient.Parent = object.gradientBar

	object.empty = Instance.new("Frame")
	object.empty.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
	object.empty.BackgroundTransparency = 0.25
	object.empty.BorderSizePixel = 0
	object.empty.Size = UDim2.fromScale(1, 0)
	object.empty.Position = UDim2.fromScale(0, 1)
	object.empty.AnchorPoint = Vector2.new(0, 1)
	object.empty.ZIndex = 22
	object.empty.Parent = object.background

	object.stroke = Instance.new("UIStroke")
	object.stroke.Thickness = HEALTH_CONFIG.BorderThickness
	object.stroke.Color = HEALTH_CONFIG.OutlineWhite
	object.stroke.Parent = object.background

	object.strokeGradient = Instance.new("UIGradient")
	object.strokeGradient.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, HEALTH_CONFIG.OutlineWhite),
		ColorSequenceKeypoint.new(0.5, HEALTH_CONFIG.OutlineBlack),
		ColorSequenceKeypoint.new(1, HEALTH_CONFIG.OutlineWhite),
	})
	object.strokeGradient.Rotation = 90
	object.strokeGradient.Parent = object.stroke

	object.displayedHealth = 1
	object.targetHealth = 1
	object.lastHealth = nil
	object.damageAnimation = nil
	object.damageConnection = nil

	HealthESPObjects[player] = object
	return object
end

local function clearDamageAnimation(object)
	if object.damageConnection then
		object.damageConnection:Disconnect()
		object.damageConnection = nil
	end

	if object.damageAnimation then
		object.damageAnimation:Destroy()
		object.damageAnimation = nil
	end
end

local function hideHealthESP(player)
	local object = HealthESPObjects[player]
	if object then
		object.container.Visible = false
		object.glow.Visible = false
	end
end

local function removeHealthESP(player)
	local object = HealthESPObjects[player]
	if not object then return end

	clearDamageAnimation(object)

	if object.container then object.container:Destroy() end
	if object.glow then object.glow:Destroy() end

	HealthESPObjects[player] = nil
end

--------------------------------------------------
-- RENDER & UPDATES
--------------------------------------------------

local function updateHealthESP(player, minX, minY, maxX, maxY, actualHealth, maxHealth, scale)
	local object = HealthESPObjects[player]
	if not object then return end

	if maxHealth <= 0 then
		hideHealthESP(player)
		return
	end

	local targetHealth = math.clamp(actualHealth / maxHealth, 0, 1)
	object.targetHealth = targetHealth

	if object.lastHealth == nil then
		object.lastHealth = targetHealth
		object.displayedHealth = targetHealth
	end

	if targetHealth < object.lastHealth then
		clearDamageAnimation(object)

		local animValue = Instance.new("NumberValue")
		animValue.Value = object.displayedHealth
		object.damageAnimation = animValue

		object.damageConnection = animValue:GetPropertyChangedSignal("Value"):Connect(function()
			object.displayedHealth = animValue.Value
		end)

		local tween = TweenService:Create(
			animValue,
			TweenInfo.new(HEALTH_CONFIG.DamageDropTime, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
			{ Value = targetHealth }
		)

		tween.Completed:Connect(function()
			if object.damageAnimation == animValue then
				object.displayedHealth = animValue.Value
				clearDamageAnimation(object)
			end
		end)

		tween:Play()
	elseif targetHealth > object.lastHealth then
		clearDamageAnimation(object)
		object.displayedHealth = targetHealth
	end

	object.lastHealth = targetHealth

	local width = HEALTH_CONFIG.Width * scale
	local gap = HEALTH_CONFIG.Gap * scale
	local boxHeight = maxY - minY
	local x = minX - gap - width
	local y = minY

	object.container.Position = UDim2.fromOffset(x, y)
	object.container.Size = UDim2.fromOffset(width, boxHeight)

	object.glow.Position = UDim2.fromOffset(x - 2 * scale, y - 2 * scale)
	object.glow.Size = UDim2.fromOffset(width + 4 * scale, boxHeight + 4 * scale)

	object.empty.Size = UDim2.fromScale(1, 1 - object.displayedHealth)
	object.strokeGradient.Rotation = getOutlineRotation()

	object.container.Visible = true
	object.glow.Visible = HEALTH_CONFIG.GlowEnabled
end

--------------------------------------------------
-- SETUP & EVENT HANDLERS
--------------------------------------------------

local function setupPlayer(player)
	if player == LocalPlayer then return end
	createHealthESP(player)

	player.CharacterAdded:Connect(function()
		task.wait(0.15)
		local health = HealthESPObjects[player] or createHealthESP(player)
		clearDamageAnimation(health)
		health.lastHealth = nil
		health.targetHealth = 1
		health.displayedHealth = 1
	end)
end

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(removeHealthESP)

RunService.RenderStepped:Connect(function()
	Camera = workspace.CurrentCamera
	if not Camera or not Enabled then return end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			local data = getCharacterData(player)
			if data then
				updateHealthESP(
					player,
					data.minX,
					data.minY,
					data.maxX,
					data.maxY,
					data.humanoid.Health,
					data.humanoid.MaxHealth,
					data.scale
				)
			else
				hideHealthESP(player)
			end
		end
	end
end)
