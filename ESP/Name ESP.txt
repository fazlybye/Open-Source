local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

--------------------------------------------------
-- CONFIGURATION
--------------------------------------------------

local NAME_CONFIG = {
	Font = 2, -- Drawing Font enum/index (2 = Monospace)

	Color = Color3.fromRGB(255, 255, 255),
	UsernameColor = Color3.fromRGB(180, 180, 180),

	Outline = true,
	OutlineColor = Color3.fromRGB(0, 0, 0),

	-- Size follows the player's on-screen height
	HeightRatio = 0.12,    -- Text size = on-screen height * this
	MaxSize = 20,          -- Largest text size (close)
	MinSize = 6,           -- Smallest text size (far)
	UsernameSizeOffset = 2,
	UsernameMinSize = 5,

	-- Hide names past this distance (0 = never hide)
	MaxDistance = 600,

	-- Fade as names get far
	FarDistance = 450,
	FadeEnabled = true,
	FadeStart = 200,
	FadeMinOpacity = 0.5,

	-- Smoothing (0 = instant, higher = smoother transition)
	Smoothing = 0.25,

	Gap = 4,
	UsernameGap = 1,
}

--------------------------------------------------
-- STORAGE & STATE
--------------------------------------------------

local NameObjects = {}
local Enabled = true

--------------------------------------------------
-- DRAWING HELPERS
--------------------------------------------------

local function createDrawingText()
	local text = Drawing.new("Text")
	text.Center = true
	text.Font = NAME_CONFIG.Font
	text.Color = Color3.fromRGB(255, 255, 255)
	text.Outline = NAME_CONFIG.Outline
	text.OutlineColor = NAME_CONFIG.OutlineColor
	text.Visible = false
	return text
end

local function createNameESP(player)
	if NameObjects[player] then
		return NameObjects[player]
	end

	local object = {
		DisplayNameTag = createDrawingText(),
		UsernameTag = createDrawingText(),
		NameSize = nil,
	}

	NameObjects[player] = object
	return object
end

local function hideNames(object)
	object.NameSize = nil
	object.DisplayNameTag.Visible = false
	object.UsernameTag.Visible = false
end

local function removeNameESP(player)
	local object = NameObjects[player]
	if not object then return end

	pcall(function() object.DisplayNameTag:Remove() end)
	pcall(function() object.UsernameTag:Remove() end)

	NameObjects[player] = nil
end

--------------------------------------------------
-- BOUNDS & CALCULATIONS
--------------------------------------------------

local function getBounds(character)
	local cf, size = character:GetBoundingBox()
	local half = size / 2

	local points = {
		cf * Vector3.new(-half.X, -half.Y, -half.Z),
		cf * Vector3.new(-half.X, -half.Y, half.Z),
		cf * Vector3.new(-half.X, half.Y, -half.Z),
		cf * Vector3.new(-half.X, half.Y, half.Z),
		cf * Vector3.new(half.X, -half.Y, -half.Z),
		cf * Vector3.new(half.X, -half.Y, half.Z),
		cf * Vector3.new(half.X, half.Y, -half.Z),
		cf * Vector3.new(half.X, half.Y, half.Z)
	}

	local minX, minY = math.huge, math.huge
	local maxX, maxY = -math.huge, -math.huge
	local visible = false

	for _, point in ipairs(points) do
		local screenPosition, onScreen = Camera:WorldToViewportPoint(point)

		if screenPosition.Z <= 0 then
			return nil
		end

		if onScreen then
			visible = true
		end

		minX = math.min(minX, screenPosition.X)
		minY = math.min(minY, screenPosition.Y)
		maxX = math.max(maxX, screenPosition.X)
		maxY = math.max(maxY, screenPosition.Y)
	end

	if not visible then
		return nil
	end

	return minX, minY, maxX, maxY
end

--------------------------------------------------
-- MAIN RENDER LOOP
--------------------------------------------------

local function updateNameESP(player)
	local object = NameObjects[player]
	if not object then return end

	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")

	if not character or not humanoid or humanoid.Health <= 0 or not root then
		hideNames(object)
		return
	end

	local distance = (Camera.CFrame.Position - root.Position).Magnitude
	local tooFar = NAME_CONFIG.MaxDistance > 0 and distance > NAME_CONFIG.MaxDistance

	if tooFar or not Enabled then
		hideNames(object)
		return
	end

	local minX, minY, maxX, maxY = getBounds(character)
	if not minX then
		hideNames(object)
		return
	end

	local height = maxY - minY
	local centerX = (minX + maxX) * 0.5

	-- Scale font size dynamically with on-screen bounding box height
	local targetSize = math.clamp(
		height * NAME_CONFIG.HeightRatio,
		NAME_CONFIG.MinSize,
		NAME_CONFIG.MaxSize
	)

	-- Apply size smoothing to remove flicker/jitter while moving
	object.NameSize = object.NameSize or targetSize
	object.NameSize = object.NameSize + (targetSize - object.NameSize) * (1 - NAME_CONFIG.Smoothing)

	local textSize = math.max(
		NAME_CONFIG.MinSize,
		math.floor(object.NameSize + 0.5)
	)

	local usernameSize = math.max(
		textSize - NAME_CONFIG.UsernameSizeOffset,
		NAME_CONFIG.UsernameMinSize
	)

	-- Adjust element spacing proportional to the current text scaling
	local sizeRatio = textSize / NAME_CONFIG.MaxSize
	local gap = math.max(2, math.floor(NAME_CONFIG.Gap * sizeRatio + 0.5))
	local lineGap = math.max(1, math.floor(NAME_CONFIG.UsernameGap * sizeRatio + 0.5))

	-- Calculate distance-based alpha attenuation
	local opacity = 1
	if NAME_CONFIG.FadeEnabled and distance > NAME_CONFIG.FadeStart then
		local f = math.clamp(
			(distance - NAME_CONFIG.FadeStart) / math.max(NAME_CONFIG.FarDistance - NAME_CONFIG.FadeStart, 1),
			0,
			1
		)
		opacity = 1 - f * (1 - NAME_CONFIG.FadeMinOpacity)
	end

	local totalHeight = textSize + lineGap + usernameSize
	local nameY = minY - totalHeight - gap
	local usernameY = nameY + textSize + lineGap

	local function applyText(tag, text, size, color, y)
		tag.Text = text
		tag.Size = size
		tag.Font = NAME_CONFIG.Font
		tag.Center = true
		tag.Color = color
		tag.Outline = NAME_CONFIG.Outline
		tag.OutlineColor = NAME_CONFIG.OutlineColor
		tag.Transparency = opacity

		-- Round positions to whole screen pixels to keep text sharp
		tag.Position = Vector2.new(
			math.floor(centerX + 0.5),
			math.floor(y + 0.5)
		)

		tag.Visible = true
	end

	applyText(
		object.DisplayNameTag,
		player.DisplayName,
		textSize,
		NAME_CONFIG.Color,
		nameY
	)

	applyText(
		object.UsernameTag,
		"@" .. player.Name,
		usernameSize,
		NAME_CONFIG.UsernameColor,
		usernameY
	)
end

--------------------------------------------------
-- SETUP & LISTENERS
--------------------------------------------------

local function setupPlayer(player)
	if player == LocalPlayer then return end
	createNameESP(player)

	player.CharacterAdded:Connect(function()
		task.wait(0.15)
		local object = NameObjects[player]
		if object then
			object.NameSize = nil
		end
	end)
end

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(removeNameESP)

RunService.RenderStepped:Connect(function()
	Camera = workspace.CurrentCamera
	if not Camera then return end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			updateNameESP(player)
		end
	end
end)
