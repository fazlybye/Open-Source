local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

--------------------------------------------------
-- CONFIGURATION
--------------------------------------------------

local HIGHLIGHT_CONFIG = {
	FillColor = Color3.fromRGB(0, 0, 0),
	OutlineColor = Color3.fromRGB(255, 255, 255),
	FillTransparency = 0.65,
	OutlineTransparency = 0,
	DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
	MaxDistance = 2000, -- Maximum view distance (0 = infinite)
}

--------------------------------------------------
-- STORAGE & STATE
--------------------------------------------------

local HighlightObjects = {}
local Enabled = true

--------------------------------------------------
-- HIGHLIGHT CREATION & CLEANUP
--------------------------------------------------

local function createHighlightESP(player)
	if HighlightObjects[player] then
		return HighlightObjects[player]
	end

	local highlight = Instance.new("Highlight")
	highlight.Name = "RunTimeHighlight_" .. player.Name
	highlight.FillColor = HIGHLIGHT_CONFIG.FillColor
	highlight.OutlineColor = HIGHLIGHT_CONFIG.OutlineColor
	highlight.FillTransparency = HIGHLIGHT_CONFIG.FillTransparency
	highlight.OutlineTransparency = HIGHLIGHT_CONFIG.OutlineTransparency
	highlight.DepthMode = HIGHLIGHT_CONFIG.DepthMode
	highlight.Enabled = false
	highlight.Parent = game:GetService("CoreGui")

	HighlightObjects[player] = highlight
	return highlight
end

local function removeHighlightESP(player)
	local highlight = HighlightObjects[player]
	if not highlight then return end

	highlight:Destroy()
	HighlightObjects[player] = nil
end

--------------------------------------------------
-- MAIN RENDER LOOP
--------------------------------------------------

local function updateHighlightESP(player)
	local highlight = HighlightObjects[player]
	if not highlight then return end

	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")

	if not character or not humanoid or humanoid.Health <= 0 or not root then
		highlight.Enabled = false
		highlight.Adornee = nil
		return
	end

	if not Enabled then
		highlight.Enabled = false
		return
	end

	-- Distance validation check
	if HIGHLIGHT_CONFIG.MaxDistance > 0 then
		local distance = (Camera.CFrame.Position - root.Position).Magnitude
		if distance > HIGHLIGHT_CONFIG.MaxDistance then
			highlight.Enabled = false
			return
		end
	end

	-- Attach highlight directly to the character model
	if highlight.Adornee ~= character then
		highlight.Adornee = character
	end

	highlight.FillColor = HIGHLIGHT_CONFIG.FillColor
	highlight.OutlineColor = HIGHLIGHT_CONFIG.OutlineColor
	highlight.FillTransparency = HIGHLIGHT_CONFIG.FillTransparency
	highlight.OutlineTransparency = HIGHLIGHT_CONFIG.OutlineTransparency
	highlight.Enabled = true
end

--------------------------------------------------
-- SETUP & LISTENERS
--------------------------------------------------

local function setupPlayer(player)
	if player == LocalPlayer then return end
	createHighlightESP(player)

	player.CharacterAdded:Connect(function()
		task.wait(0.15)
		local highlight = HighlightObjects[player]
		if highlight then
			highlight.Enabled = false
			highlight.Adornee = nil
		end
	end)
end

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(removeHighlightESP)

RunService.RenderStepped:Connect(function()
	Camera = workspace.CurrentCamera
	if not Camera then return end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			updateHighlightESP(player)
		end
	end
end)
