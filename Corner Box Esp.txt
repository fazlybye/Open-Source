local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

local CONFIG = {
	MainColor = Color3.fromRGB(0, 0, 0),
	OutlineColor = Color3.fromRGB(255, 255, 255),
	GlowColor = Color3.fromRGB(0, 0, 0),

	CornerSize = 0.20,

	MainThickness = 2,
	OutlineThickness = 5,

	GlowEnabled = true,
	GlowLayers = 4,
	GlowSpread = 22,

	OutlineTransparency = 0.05,
	MainTransparency = 0,

	Pulse = true,
	PulseSpeed = 2,
	PulseAmount = 0.10,

	MaxDistance = 2000,

	FullSizeBox = 45,
	MinScale = 0.3,

	ImageId = "rbxassetid://138773730625079",
	ImageTransparency = 0.3,
	ImageSize = 0.38,
	ImageMaxRatio = 0.42,
	ImageMinSize = 3,
	ImageMaxSize = 68,
	ImagePadding = 2,
	ImageFadeStart = 10,
	ImageFadeEnd = 40
}

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "ModernHUD_ESP"
ScreenGui.IgnoreGuiInset = true
ScreenGui.ResetOnSpawn = false
ScreenGui.DisplayOrder = 999999
ScreenGui.Parent = game:GetService("CoreGui")

local espObjects = {}

local function round(value)
	return math.floor(value + 0.5)
end

local function createLine(parent, zIndex, color, transparency)
	local line = Instance.new("Frame")
	line.AnchorPoint = Vector2.new(0.5, 0.5)
	line.BackgroundColor3 = color
	line.BackgroundTransparency = transparency
	line.BorderSizePixel = 0
	line.ZIndex = zIndex
	line.Size = UDim2.fromOffset(1, 1)
	line.Parent = parent

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = line

	return line
end

local function createSegment(parent, horizontal)
	local segment = {
		horizontal = horizontal,
		glow = {},
		outline = nil,
		main = nil
	}

	if CONFIG.GlowEnabled then
		for i = CONFIG.GlowLayers, 1, -1 do
			local alpha = 0.82 + (i * 0.035)

			segment.glow[i] = createLine(
				parent,
				1,
				CONFIG.GlowColor,
				math.clamp(alpha, 0, 0.97)
			)
		end
	end

	segment.outline = createLine(
		parent,
		10,
		CONFIG.OutlineColor,
		CONFIG.OutlineTransparency
	)

	segment.main = createLine(
		parent,
		11,
		CONFIG.MainColor,
		CONFIG.MainTransparency
	)

	return segment
end

local function createCorner(parent)
	return {
		horizontal = createSegment(parent, true),
		vertical = createSegment(parent, false)
	}
end

local function createImage(parent)
	local image = Instance.new("ImageLabel")
	image.AnchorPoint = Vector2.new(0, 0)
	image.BackgroundTransparency = 1
	image.BorderSizePixel = 0
	image.Image = CONFIG.ImageId
	image.ImageTransparency = CONFIG.ImageTransparency
	image.ScaleType = Enum.ScaleType.Fit
	image.ZIndex = 12
	image.Size = UDim2.fromOffset(1, 1)
	image.Parent = parent

	return image
end

local function createESP(player)
	if espObjects[player] then
		return
	end

	local container = Instance.new("Frame")
	container.Name = player.Name
	container.BackgroundTransparency = 1
	container.BorderSizePixel = 0
	container.Size = UDim2.fromScale(1, 1)
	container.Parent = ScreenGui

	espObjects[player] = {
		container = container,
		topLeft = createCorner(container),
		topRight = createCorner(container),
		bottomLeft = createCorner(container),
		bottomRight = createCorner(container),
		image = createImage(container)
	}
end

local function removeESP(player)
	local esp = espObjects[player]

	if esp then
		esp.container:Destroy()
		espObjects[player] = nil
	end
end

local function setSegment(segment, x, y, length, pulse, scale)
	x = round(x)
	y = round(y)
	length = math.max(2, round(length))

	local baseThickness = math.max(1, round(CONFIG.MainThickness * scale))
	local mainThickness = baseThickness

	if baseThickness >= 2 then
		mainThickness = math.max(2, round(baseThickness * pulse))
	end

	local outlinePad = math.max(1, round(((CONFIG.OutlineThickness - CONFIG.MainThickness) / 2) * scale))
	local outlineThickness = baseThickness + outlinePad * 2

	local mainSize
	local outlineSize

	if segment.horizontal then
		mainSize = UDim2.fromOffset(length, mainThickness)
		outlineSize = UDim2.fromOffset(length + outlinePad * 2, outlineThickness)
	else
		mainSize = UDim2.fromOffset(mainThickness, length)
		outlineSize = UDim2.fromOffset(outlineThickness, length + outlinePad * 2)
	end

	segment.main.Position = UDim2.fromOffset(x, y)
	segment.main.Size = mainSize

	segment.outline.Position = UDim2.fromOffset(x, y)
	segment.outline.Size = outlineSize

	for i, glow in ipairs(segment.glow) do
		local spread = round(CONFIG.GlowSpread * scale * (i / CONFIG.GlowLayers))

		glow.Position = UDim2.fromOffset(x, y)

		if segment.horizontal then
			glow.Size = UDim2.fromOffset(
				length + spread,
				baseThickness + spread
			)
		else
			glow.Size = UDim2.fromOffset(
				baseThickness + spread,
				length + spread
			)
		end
	end
end

local function setCorner(
	corner,
	x,
	y,
	horizontalLength,
	verticalLength,
	horizontalDirection,
	verticalDirection,
	pulse,
	scale
)
	setSegment(
		corner.horizontal,
		x + (horizontalLength / 2) * horizontalDirection,
		y,
		horizontalLength,
		pulse,
		scale
	)

	setSegment(
		corner.vertical,
		x,
		y + (verticalLength / 2) * verticalDirection,
		verticalLength,
		pulse,
		scale
	)
end

local function setImage(image, minX, minY, width, height, scale)
	if CONFIG.ImageId == "" or CONFIG.ImageId == "rbxassetid://0" then
		image.Visible = false
		return
	end

	local smallestSide = math.min(width, height)

	local baseThickness = math.max(1, round(CONFIG.MainThickness * scale))
	local outlinePad = math.max(1, round(((CONFIG.OutlineThickness - CONFIG.MainThickness) / 2) * scale))
	local padding = math.floor(CONFIG.ImagePadding * scale)

	local fade = math.clamp(
		(smallestSide - CONFIG.ImageFadeStart) / (CONFIG.ImageFadeEnd - CONFIG.ImageFadeStart),
		0,
		1
	)

	local offset = math.ceil(baseThickness / 2) + round((outlinePad + padding) * fade)

	local size = round(math.clamp(
		smallestSide * CONFIG.ImageSize,
		CONFIG.ImageMinSize,
		CONFIG.ImageMaxSize
	))

	size = math.min(size, math.max(2, round(smallestSide * CONFIG.ImageMaxRatio)))

	image.Position = UDim2.fromOffset(round(minX) + offset, round(minY) + offset)
	image.Size = UDim2.fromOffset(size, size)
	image.Visible = true
end

local function getBounds(character)
	local cf, size = character:GetBoundingBox()
	local half = size / 2

	local points = {
		cf * Vector3.new(-half.X, -half.Y, -half.Z),
		cf * Vector3.new(-half.X, -half.Y, half.Z),
		cf * Vector3.new(-half.X, half.Y, -half.Z),
		cf * Vector3.new(-half.X, half.Y, half.Z),
		cf * Vector3.new(half.X, -half.Y, -half.Z),
		cf * Vector3.new(half.X, -half.Y, half.Z),
		cf * Vector3.new(half.X, half.Y, -half.Z),
		cf * Vector3.new(half.X, half.Y, half.Z)
	}

	local minX = math.huge
	local minY = math.huge
	local maxX = -math.huge
	local maxY = -math.huge
	local visible = false

	for _, point in ipairs(points) do
		local screenPosition, onScreen = Camera:WorldToViewportPoint(point)

		if screenPosition.Z <= 0 then
			return nil
		end

		if onScreen then
			visible = true
		end

		minX = math.min(minX, screenPosition.X)
		minY = math.min(minY, screenPosition.Y)
		maxX = math.max(maxX, screenPosition.X)
		maxY = math.max(maxY, screenPosition.Y)
	end

	if not visible then
		return nil
	end

	return minX, minY, maxX, maxY
end

local function updateESP(player, esp, time)
	local character = player.Character

	if not character then
		esp.container.Visible = false
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")

	if not humanoid or humanoid.Health <= 0 or not root then
		esp.container.Visible = false
		return
	end

	local distance = (Camera.CFrame.Position - root.Position).Magnitude

	if distance > CONFIG.MaxDistance then
		esp.container.Visible = false
		return
	end

	local minX, minY, maxX, maxY = getBounds(character)

	if not minX then
		esp.container.Visible = false
		return
	end

	local width = maxX - minX
	local height = maxY - minY

	if width < 3 or height < 3 then
		esp.container.Visible = false
		return
	end

	esp.container.Visible = true

	local pulse = 1

	if CONFIG.Pulse then
		pulse = 1 + math.sin(time * CONFIG.PulseSpeed) * CONFIG.PulseAmount
	end

	local scale = math.clamp(math.min(width, height) / CONFIG.FullSizeBox, CONFIG.MinScale, 1)

	local cornerLength = math.min(width, height) * CONFIG.CornerSize
	cornerLength = math.clamp(cornerLength, 3, 50)
	cornerLength = math.min(cornerLength, width * 0.3, height * 0.3)

	setCorner(
		esp.topLeft,
		minX,
		minY,
		cornerLength,
		cornerLength,
		1,
		1,
		pulse,
		scale
	)

	setCorner(
		esp.topRight,
		maxX,
		minY,
		cornerLength,
		cornerLength,
		-1,
		1,
		pulse,
		scale
	)

	setCorner(
		esp.bottomLeft,
		minX,
		maxY,
		cornerLength,
		cornerLength,
		1,
		-1,
		pulse,
		scale
	)

	setCorner(
		esp.bottomRight,
		maxX,
		maxY,
		cornerLength,
		cornerLength,
		-1,
		-1,
		pulse,
		scale
	)

	setImage(esp.image, minX, minY, width, height, scale)
end

local function setupPlayer(player)
	if player == LocalPlayer then
		return
	end

	createESP(player)

	player.CharacterAdded:Connect(function()
		task.wait(0.15)

		if not espObjects[player] then
			createESP(player)
		end
	end)
end

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

Players.PlayerAdded:Connect(setupPlayer)

Players.PlayerRemoving:Connect(function(player)
	removeESP(player)
end)

RunService.RenderStepped:Connect(function()
	Camera = workspace.CurrentCamera

	if not Camera then
		return
	end

	local time = os.clock()

	for player, esp in pairs(espObjects) do
		updateESP(player, esp, time)
	end
end)
